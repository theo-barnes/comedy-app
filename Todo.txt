## Supabase Security Patches (run in Supabase SQL editor)

-- PATCH 1: Prevent role escalation
-- The current UPDATE policy lets users change their own role (fan → comedian/venue).
-- Replace it with one that locks the role column after creation.

drop policy "Users can update own profile" on public.profiles;

create policy "Users can update own profile"
  on public.profiles for update
  using (auth.uid() = id)
  with check (
    auth.uid() = id
    and role = (select p.role from public.profiles p where p.id = auth.uid())
  );

-- PATCH 2: Auto-update updated_at on every profile change

create or replace function public.handle_updated_at()
returns trigger language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger on_profile_updated
  before update on public.profiles
  for each row execute procedure public.handle_updated_at();
